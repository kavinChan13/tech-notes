#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <cstring>
#include <iostream>

int main() {
    // 1. 创建 socket (IPv4, TCP)
    int listenfd = socket(AF_INET, SOCK_STREAM, 0);

    // 2. 绑定地址和端口
    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY);  // 监听所有网卡
    addr.sin_port = htons(8080);               // 端口 8080

    // SO_REUSEADDR: 允许快速重启时复用端口
    int opt = 1;
    setsockopt(listenfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    bind(listenfd, (sockaddr*)&addr, sizeof(addr));

    // 3. 开始监听，backlog = 128
    listen(listenfd, 128);
    std::cout << "Server listening on port 8080...\n";

    // 4. 接受连接 (阻塞)
    sockaddr_in client_addr{};
    socklen_t client_len = sizeof(client_addr);
    int connfd = accept(listenfd, (sockaddr*)&client_addr, &client_len);

    // 5. 读写数据
    char buf[1024];
    ssize_t n = read(connfd, buf, sizeof(buf));
    buf[n] = '\0';
    std::cout << "Received: " << buf << "\n";

    const char* resp = "Hello from server!";
    write(connfd, resp, strlen(resp));

    // 6. 关闭
    close(connfd);
    close(listenfd);
}

2.3 socket 地址结构

字节序（Endianness）：多字节整数在内存里从低地址到高地址的排列方式。大端把高字节放在低地址（像从左到右写数字）；小端把低字节放在低地址。x86 / 主流 ARM 是小端，但 TCP/IP 规定网络字节序 = 大端，所以 sin_port、sin_addr 以及自定义协议里的长度字段，发送前要用 htons / htonl 转成网络序，接收后再 ntohs / ntohl 转回主机序。单机内存读写不必关心端序；跨主机、跨架构或落盘二进制格式时才必须统一。

// 发送端: 先发长度，再发数据
bool sendMessage(int fd, const std::string& msg) {
    uint32_t len = htonl(msg.size());  // 网络字节序!
    // 发送 4 字节长度头
    if (write(fd, &len, 4) != 4) return false;
    // 发送消息体
    if (write(fd, msg.data(), msg.size()) != (ssize_t)msg.size()) return false;
    return true;
}

// 接收端: 先读长度，再读精确数量的字节
std::string recvMessage(int fd) {
    // 1. 读取 4 字节长度头
    uint32_t netLen;
    readExact(fd, &netLen, 4);         // 必须精确读满 4 字节
    uint32_t len = ntohl(netLen);

    // 2. 根据长度读取消息体
    std::string msg(len, '\0');
    readExact(fd, msg.data(), len);    // 必须精确读满 len 字节
    return msg;
}

// 关键辅助函数: 确保读满 n 字节 (处理拆包)
void readExact(int fd, void* buf, size_t n) {
    size_t totalRead = 0;
    while (totalRead < n) {
        ssize_t r = read(fd, (char*)buf + totalRead, n - totalRead);
        if (r <= 0) throw std::runtime_error("connection closed");
        totalRead += r;
    }
}

#include <sys/epoll.h>

// 1. 创建 epoll 实例
int epfd = epoll_create1(0);

// 2. 注册感兴趣的 fd 和事件
struct epoll_event ev;
ev.events = EPOLLIN | EPOLLET;  // 读事件 + 边缘触发
ev.data.fd = listenfd;
epoll_ctl(epfd, EPOLL_CTL_ADD, listenfd, &ev);

// 3. 事件循环
struct epoll_event events[1024];
while (true) {
    int n = epoll_wait(epfd, events, 1024, -1);  // 阻塞等待

    for (int i = 0; i < n; ++i) {
        if (events[i].data.fd == listenfd) {
            // 新连接到来
            int connfd = accept(listenfd, ...);
            // 设置非阻塞 + 注册到 epoll
            setNonBlocking(connfd);
            ev.events = EPOLLIN | EPOLLET;
            ev.data.fd = connfd;
            epoll_ctl(epfd, EPOLL_CTL_ADD, connfd, &ev);
        } else {
            // 数据到来，处理读写
            handleConnection(events[i].data.fd);
        }
    }
}

6.2 sendfile 代码

#include <sys/sendfile.h>
#include <fcntl.h>

// 传统方式: read + write (4次拷贝)
char buf[8192];
while ((ssize_t n = read(filefd, buf, sizeof(buf))) > 0) {
    write(sockfd, buf, n);  // 数据经过用户空间中转
}

// sendfile 方式: 零拷贝 (2次DMA拷贝)
struct stat st;
fstat(filefd, &st);
sendfile(sockfd,    // 目标: socket fd
         filefd,    // 源: 文件 fd
         nullptr,  // offset (nullptr = 从当前位置)
         st.st_size // 发送的字节数
);
// 数据直接从文件 → 内核 → 网卡，不经过用户空间!

7b.2 QUIC —— TCP 的"用户态 + UDP"替代

QUIC（RFC 9000，原 Google 2012）跑在 UDP 上，把可靠传输 + 拥塞控制 + 加密都搬到用户态。终结 TCP/TLS 的几大顽疾。

QUIC 解决了什么

7c.2 SO_REUSEPORT —— 多进程/多线程监听同一端口

Linux 3.9+ 的杀手特性。允许多个进程/线程同时 bind 同一 (IP, port)，内核做4 元组哈希分流到各 listener，彻底解决 nginx 旧版 "thundering herd" 问题。

替代旧的 master-fork-worker 模式

// 每个 worker 进程 / 线程独立 socket + bind
for (int i = 0; i < N_WORKERS; ++i) {
    if (fork() == 0) {
        int sock = socket(AF_INET, SOCK_STREAM, 0);
        int opt = 1;
        setsockopt(sock, SOL_SOCKET, SO_REUSEPORT, &opt, sizeof(opt));   // ★
        bind(sock, ...);
        listen(sock, 128);
        worker_loop(sock);     // 内核自动分流连接
        break;
    }
}

7c.3 TCP FastOpen — 把数据塞进握手包

普通 TCP 必须先 3 次握手才能发数据，TFO（RFC 7413）允许客户端在 SYN 包里就携带应用数据，节省 1 RTT。

// 服务端启用
int qlen = 5;
setsockopt(sock, IPPROTO_TCP, TCP_FASTOPEN, &qlen, sizeof(qlen));

// 客户端发起：sendto 替代 connect+send
sendto(sock, data, len, MSG_FASTOPEN, (sockaddr*)&addr, sizeof(addr));

需要：① Linux 3.7+；② 双端都支持；③ HTTP/2 / gRPC 等长连接收益小，短连接（DNS / HTTP/1.0）收益大。

7c.4 TCP_NODELAY / TCP_CORK / TCP_QUICKACK 三件套

option作用用在哪
TCP_NODELAY禁用 Nagle 算法（小包不缓存等大包合并）低延时 / RPC / 交互式
TCP_CORK反 Nagle —— 强制缓存直到 cork 解除或 buffer 满批量发大块（如 sendfile + header）
TCP_QUICKACK禁用延迟 ACK（200ms 等待）请求-响应短交互
常见坑：Nagle + delayed ACK 互相等待 → 出现 200ms "卡顿"。一般 TCP_NODELAY 总该开（除非你确实想合并小包）。

7c.5 SO_BUSY_POLL — 用户态主动轮询

把 NIC 中断 → 内核 → 应用通知的"等中断"流程改成应用主动 poll，消灭中断延迟。延迟敏感场景能从 ~10 μs 降到 ~3 μs。

// 应用 socket 上启用

// 也可以全局
sysctl -w net.core.busy_poll=50
sysctl -w net.core.busy_read=50

代价：吃 CPU 高（轮询时本核 100%）。HFT / 低延时 RPC 用，普通业务不要开。

点击每道题展开参考答案（先自测，再对照）。

  1. 基础为什么 TCP 握手是三次而不是两次？

    因为握手的目的是让双方都确认对方的收发能力，并同步初始序列号（ISN）。两次握手只能让客户端确认"服务端收到了我的 SYN"，服务端却无法确认自己的 SYN-ACK 是否到达——它的 ISN 没被确认，连接是半成品。

    更具体的危害：一个早已被客户端放弃的旧 SYN 在网络里绕了一圈后延迟到达，两次握手下服务端会直接建立连接并分配缓冲区，而客户端根本不认这条连接，于是服务端资源白白挂着。三次握手时客户端会对这个意外的 SYN-ACK 回 RST，连接不会建立。

    顺带说清一个常被追问的点：ISN 不是从 0 开始，而是随时间递增 + 加密哈希生成的，既避免旧连接的残包被新连接误收，也防止盲序列号猜测攻击。

  2. 基础为什么挥手要四次而不是三次？

    因为 TCP 是全双工的，两个方向要各自关闭一次。收到 FIN 只说明"对方不再发数据了"，本端可能还有数据没发完，所以 ACK（我知道你说完了）和自己的 FIN（我也说完了）必须分开，中间还能继续发数据——这就是半关闭状态（shutdown(fd, SHUT_WR) 的用途）。

    那什么时候会变成"三次"？当被动方收到 FIN 时确实没有数据要发，内核可以把 ACK 和 FIN 合并成一个包（延迟 ACK 机制给了它这个机会）。所以"四次"是协议允许的最大次数，抓包看到三次也完全正常。

    工程意义：如果服务端出现大量 CLOSE_WAIT，说明它收到了 FIN 并回了 ACK，但迟迟没有调用 close() 发出自己的 FIN——这几乎总是应用层的 bug（read() 返回 0 后忘了关 fd），不是内核参数能解决的。

  3. 基础TIME_WAIT 状态为什么要等 2MSL？

    MSL（Maximum Segment Lifetime）是一个报文在网络中存活的最长时间，Linux 取 30 s，所以 2MSL = 60 s（net.ipv4.tcp_fin_timeout 调的其实是 FIN_WAIT_2，不是这个）。等两倍 MSL 有两个理由：

    • 保证最后那个 ACK 能送达：如果它丢了，对端会重传 FIN。主动关闭方必须还"活着"才能重发 ACK，否则对端收到的是 RST，会误判为异常关闭。一去一回正好是 2 个 MSL。
    • 让旧连接的残包彻底消散：否则同一个四元组的新连接可能收到上一条连接的迟到数据，造成数据串流。

    生产上 TIME_WAIT 堆积（几万条）通常是短连接 + 客户端主动关闭造成的，会耗尽本地端口（ip_local_port_range）导致 EADDRNOTAVAIL。正确的处理顺序是：① 改长连接 / 连接池，这是治本；② 服务端加 SO_REUSEADDR 解决重启时的 EADDRINUSE；③ 客户端侧开 net.ipv4.tcp_tw_reuse=1（依赖 TCP 时间戳，只对发起连接生效，是安全的）。

    ⚠️ 不要用 tcp_tw_recycle——它会对同一 NAT 出口后的多台客户机做时间戳校验，导致部分客户端随机连不上，Linux 4.12 起已经删除。这是面试里区分"背过参数"和"踩过坑"的经典分界。

  4. 中级epoll 的 ET 模式为什么必须配合非阻塞 fd？

    ET（边缘触发）只在 fd 状态发生变化时通知一次，所以拿到事件后必须一次把数据读干净，否则剩下的数据不会再触发第二次通知，连接就"卡死"了。"读干净"的判定只能靠 read() 返回 EAGAIN——而阻塞 fd 在无数据时不会返回 EAGAIN，它会阻塞在那里，整个事件循环线程被一个连接拖住，其它几万个 fd 全部停摆。

    所以 ET 的正确写法是固定套路：

    while (true) {
    ssize_t n = read(fd, buf, sizeof(buf));
    if (n > 0)  { consume(buf, n); continue; }
    if (n == 0) { close_connection(); break; }   // 对端关闭
    if (errno == EINTR)  continue;
    if (errno == EAGAIN || errno == EWOULDBLOCK) break;  // 真正读空了
    handle_error(); break;
}

    写方向同理：EPOLLOUT 在 ET 下只在"从不可写变可写"时触发一次，必须循环 write 到 EAGAIN，并且平时不要一直注册 EPOLLOUT（否则会空转），只在写缓冲区满、有数据积压时临时加上。

    选型上补一句：LT 并不比 ET 慢多少，且不容易写错（少读一点下次还会通知）。ET 的真实收益是在"一次事件里有大量数据"时减少 epoll_wait 往返次数，以及配合 EPOLLEXCLUSIVE 缓解惊群。nginx 用 ET，libevent 默认 LT，都能扛住高并发——先把正确性做对再谈这点差异。

9.1 TCP 拥塞控制 — 网络流量的平衡艺术

TCP 通过拥塞窗口 cwnd（发送端）和接收窗口 rwnd（接收端）动态调整发送速率。实际发送量 = min(cwnd, rwnd)。

9.2 Nagle + 延迟 ACK 的交互陷阱

Nagle 算法：小包（< MSS）延迟发送，累积成大包再发。
延迟 ACK：收到数据后不立即回 ACK，等 40ms 或有数据捎带。
两者叠加：请求-响应型应用会出现 ~40ms 莫名延迟。

// 关闭 Nagle — 实时协议的标配
int flag = 1;
setsockopt(fd, IPPROTO_TCP, TCP_NODELAY, &flag, sizeof(flag));

// 关闭延迟 ACK (Linux)
setsockopt(fd, IPPROTO_TCP, TCP_QUICKACK, &flag, sizeof(flag));

// 或：应用层 buffer，writev 一次发完整消息 (最优)
writev(fd, iov, 2);  // header + body 原子写

原则：请求-响应（RPC、RPC）一律 TCP_NODELAY；吞吐型（文件传输）保留 Nagle。

  1. 基础accept 返回的 fd 跟 listenfd 有什么区别？

    两者是完全不同的东西，只是都用 fd 表示。

    • listenfd：不承载数据。它绑定的是「本地 IP + 端口」这个二元组，内核在它下面挂了两个队列——半连接队列（收到 SYN、还没完成握手，长度受 tcp_max_syn_backlog 约束）和 全连接队列（握手已完成、等着被 accept() 取走，长度是 min(backlog, somaxconn)）。对它 read() 是没有意义的；它可读的含义是"全连接队列里有东西"。
    • connfd：一条已建立的连接，由完整的四元组（源 IP、源端口、目的 IP、目的端口）唯一标识，有自己独立的发送/接收缓冲区、序列号、窗口和拥塞状态。

    所以同一个 listenfd 上可以 accept 出几十万个 connfd，它们共享同一个本地端口，靠四元组区分——这也解释了为什么"一个端口只能有一个连接"是错的：服务端的连接数不受端口数限制，受 fd 上限和内存限制；受端口数限制的是客户端（同一目标地址最多约 6 万个本地端口）。

    还有一个实践细节：accept4(..., SOCK_NONBLOCK | SOCK_CLOEXEC) 比 accept() 之后再 fcntl() 少两次系统调用，且避免了多线程下 fork 泄漏 fd 的竞态——高并发服务应该默认用它。

  2. 基础TCP 和 UDP 的核心区别是什么？各自适合什么场景？

    一句话：TCP 卖的是"可靠的字节流"，UDP 卖的是"尽力而为的数据报"。所有差异都从这句话推得出来。

    维度TCPUDP
    连接三次握手建连，有状态无连接，发了就走
    边界字节流，无消息边界（要自己分包）保留数据报边界，一发一收
    可靠性确认 + 重传 + 排序 + 去重不保证到达、不保证顺序
    流控 / 拥塞控制滑动窗口 + 拥塞算法没有，可能压垮网络
    头部开销20 字节起（含选项更多）固定 8 字节
    时延特性丢包会触发重传，产生长尾丢了就丢了，时延稳定

    怎么选：需要"不能少一个字节"的走 TCP（HTTP、数据库、文件传输、RPC）；需要"迟到的数据不如不要"的走 UDP（音视频实时通话、游戏状态同步、DNS 查询、时间同步）。

    关键在于理解为什么实时音视频不用 TCP：不是因为 TCP 慢，而是因为它的重传是强制的——一个 200 ms 前的音频包重传到达时早就该播完了，却还要阻塞它后面所有已经到达的数据（队头阻塞）。UDP 把"要不要重传、重传几次"的决定权交还给应用层，这才是真正的价值。QUIC 走的也是这个思路：在 UDP 上自己实现可靠性，但按流（stream）独立处理。

  3. 基础为什么 TCP 会"粘包"？UDP 会吗？工程上怎么定义消息边界？

    "粘包"这个词其实有误导性——TCP 从来就没有包的概念，它提供的是一条无边界的字节流。你调两次 write() 发 "ABC" 和 "DE"，接收方一次 read() 读到 "ABCDE" 完全符合协议语义，不是 bug。所以正确的说法是"应用层没有定义消息边界"。

    造成边界模糊的具体机制有三层：发送端的 Nagle 算法会把小包攒起来合并；内核发送/接收缓冲区本来就是字节队列；路径 MTU 会把大消息拆成多个段。UDP 则保留数据报边界——一次 sendto() 对应一次 recvfrom()，要么完整收到要么收不到，所以 UDP 没有这个问题（但有另一个问题：超过 MTU 会 IP 分片，丢一片整个报文报废）。

    三种定边界的方案，按通用性排序：

    • 长度前缀（最通用）：包头固定 4 字节大端长度 + 变长包体。收端先读满 4 字节，解析出长度 N，再读满 N 字节。必须实现 readExact() 循环读，因为一次 read() 不保证读满。要对 N 做上限校验，否则一个恶意的 0xFFFFFFFF 就能让你 malloc 4 GB。
    • 分隔符：如 HTTP 的 \r\n\r\n、Redis RESP 的 \r\n。可读性好，但正文里出现分隔符时要转义，且要限制单行最大长度防 DoS。
    • 固定长度：只适合定长的控制报文，浪费带宽，扩展性差。

    接收端必须维护每连接一个缓冲区，把不完整的半包留到下次事件继续拼——这是所有网络框架里最容易写出 bug 的一段代码，也是面试官最爱让人手写的一段。

  4. 基础大端 vs 小端？什么时候必须关心？

    以 uint32_t x = 0x12345678 为例：

    • 大端：内存从低地址起为 12 34 56 78（高位字节在低地址）——网络字节序、MIPS / PowerPC 等。
    • 小端：内存从低地址起为 78 56 34 12（低位字节在低地址）——x86、主流 ARM。

    为什么会有两种：纯历史选型差异。小端在做低位运算和类型截断时更自然（把 uint32_t* 强转成 uint8_t* 读到的就是最低字节），大端符合人类书写顺序、调试时看内存 dump 更直观。网络协议统一约定用大端，只是为了有个共同标准，跟哪种"更好"无关。

    什么时候必须处理——凡是字节要离开本机内存的场合：

    • Socket API 的地址与端口：htons/htonl、ntohs/ntohl（在大端机上编译成 no-op，所以不写也能在某些机器上跑对，这类 bug 最阴险）。
    • 自定义二进制协议：禁止把结构体直接 memcpy 当线格式——除了端序，还有对齐填充和 long 宽度在不同平台不一致的问题。要用固定宽度类型（uint32_t）逐字段序列化。
    • 跨平台的文件格式、共享内存、嵌入式设备与 PC 互通。

    C++20 起可以用 std::endian::native 在编译期判断，std::byteswap（C++23）做转换，不必再写宏。Protobuf / FlatBuffers 等序列化库内部已经处理好；真正出事的永远是手写的"图省事"协议。

9.8.2 中级（区分熟练度）

  1. 中级TCP 靠什么做到可靠？既然如此，HTTP/3 为什么反而换回 UDP？

    TCP 的可靠性由五件事共同保证：① 序列号 + 累计确认（知道谁到了、谁没到）；② 超时重传与快速重传（RTO 由 RTT 采样动态估算，收到 3 个重复 ACK 立即重传而不等超时）；③ 滑动窗口保证顺序与去重；④ 流量控制（接收窗口，防止压垮对端）；⑤ 拥塞控制（拥塞窗口，防止压垮网络）。注意流控和拥塞控制是两回事——前者对端说了算，后者靠自己推断，实际发送量取两者较小值。

    HTTP/3 换 QUIC 的动机，核心是队头阻塞（HoL blocking）。HTTP/2 已经在应用层做了多路复用，但它跑在单条 TCP 连接上：任何一个 TCP 段丢失，内核就必须等它重传到达才能把后续字节交给应用——哪怕丢的那段属于流 A，流 B、C、D 的数据早就到齐了也得干等。应用层的多路复用被传输层的单一字节流按住了。

    QUIC 把可靠性实现搬到用户态的 UDP 之上，于是能做到：

    • 流级独立重传：一个流丢包不影响其它流交付，真正实现多路复用。
    • 连接迁移：连接由 Connection ID 标识而非四元组，Wi-Fi 切 5G 时连接不断。
    • 握手更快：传输握手与 TLS 1.3 握手合并，首次 1-RTT，恢复 0-RTT（TCP+TLS 至少要 2–3 RTT）。
    • 可快速演进：协议栈在用户态，改一版发一版；TCP 的新特性要等操作系统和全网中间设备升级，周期以十年计。

    代价要说清：UDP 在用户态收发，系统调用与内存拷贝开销比内核 TCP 大（靠 GSO/GRO、sendmmsg 缓解）；部分企业网络会限速或封禁 UDP，所以实际部署都要保留 TCP 回退路径。

  2. 中级惊群效应是什么？在 accept 和 epoll 两个层面分别怎么解决？

    惊群：多个进程/线程阻塞等待同一个事件源，事件到来时内核把它们全部唤醒，但只有一个能真正拿到工作，其余的白白经历一次上下文切换后重新睡下。核数越多、worker 越多，浪费越明显。

    要分清两个层次，它们的修复时间点差了十几年：

    • accept 惊群：多个进程直接阻塞在同一个 listenfd 的 accept() 上。Linux 2.6 起内核已改为只唤醒一个（等待队列上加 WQ_FLAG_EXCLUSIVE），这个问题本身已经不存在了。
    • epoll 惊群：多个进程把同一个 listenfd 加进各自的 epoll，阻塞在 epoll_wait() 上。这时内核唤醒的是"所有关注该 fd 的 epoll 实例"，惊群回来了。Linux 4.5+ 提供 EPOLLEXCLUSIVE 标志，注册时带上即可让内核只唤醒一个（或少数几个）。nginx 早年则是用 accept_mutex 在用户态串行化，代价是引入锁竞争和延迟。

    根治方案是 SO_REUSEPORT（Linux 3.9+）：每个 worker 创建自己的 listenfd 并绑定同一个端口，内核按四元组哈希把新连接直接投递到其中一个队列。这样根本不存在"多个等待者竞争同一队列"，还顺带解决了 accept_mutex 带来的负载不均。

    int fd = socket(AF_INET, SOCK_STREAM | SOCK_NONBLOCK, 0);
int on = 1;
setsockopt(fd, SOL_SOCKET, SO_REUSEPORT, &on, sizeof(on));  // 必须在 bind 之前
bind(fd, ...); listen(fd, 1024);

    它也有坑：内核按连接建立时的哈希分发，某个 worker 恰好接到一批长连接大户时会不均；worker 进程退出时它队列里已完成握手、还没 accept 的连接会被丢弃（客户端看到 RST），所以优雅重启需要额外处理。

  3. 中级epoll 为什么比 select/poll 高效？说说底层数据结构。

    关键差别不在"轮询 vs 事件"这句口号，而在每次调用要传递和扫描多少数据。

    • select/poll：每次调用都要把整个 fd 集合从用户态拷进内核，内核再遍历全部 fd 逐个调用其 poll 方法检查状态，返回后用户态还要再遍历一遍找出就绪的。三重 O(n)，n 是关注的 fd 数。select 还有 FD_SETSIZE=1024 的硬上限，且 fd_set 会被内核改写、每轮都得重建。
    • epoll：把"关注哪些 fd"这份状态常驻内核（红黑树，epoll_ctl 增删改 O(log n)，只在变化时付费）。注册时内核往每个 fd 的设备等待队列挂一个回调 ep_poll_callback；数据到达时驱动/协议栈触发回调，把该 epitem 塞进 epoll 实例的就绪链表（rdllist）。epoll_wait 只需搬走这条链表，复杂度是 O(就绪数) 而不是 O(总数)。

    所以 epoll 的优势随「总连接数 / 活跃连接数」的比值放大：10 万连接里只有 100 个活跃时，epoll 几乎零成本，select 要扫 10 万次。反过来，如果几乎所有 fd 每轮都活跃，epoll 相对 poll 并没有优势（还多了红黑树维护成本）——这正是面试官想听到的"知其所以然"。

    再补两个常被追问的点：① epoll 的红黑树 + 就绪链表都在内核，epoll_wait 返回的事件数组是唯一的用户-内核拷贝，且大小只与就绪数成正比；② epoll_wait 的返回值上限由你传入的 maxevents 决定，设太小会导致一轮处理不完、频繁往返，通常取 128–1024。

  4. 中级HTTPS 握手到底做了哪几件事？有哪些可落地的性能优化？

    TLS 握手同时解决三个问题，缺一不可：身份认证（证书链验证，确认对方是谁）、密钥协商（协商出只有双方知道的对称密钥）、参数协商（密码套件、协议版本、ALPN 里的 h2/h3）。真正传数据时用的是对称加密（AES-GCM / ChaCha20-Poly1305），非对称加密只用在握手阶段——因为它慢好几个数量级。

    • TLS 1.2：2-RTT。ClientHello → ServerHello + 证书 + ServerKeyExchange → ClientKeyExchange + ChangeCipherSpec → Finished。加上 TCP 的 1-RTT，首字节要等 3 个 RTT。
    • TLS 1.3：1-RTT。客户端在 ClientHello 里就猜好曲线并带上自己的 key share，服务端一个回合就能算出共享密钥。同时砍掉了 RSA 密钥传输、静态 DH、CBC 等不安全套件，强制前向保密（只保留 ECDHE）。会话恢复时可 0-RTT。

    优化手段，按收益排序：

    • 连接复用——收益最大且最容易被忽略。握手再快也不如不握手：开 HTTP/2 或长连接，把上千次请求摊到一次握手上。客户端侧务必用连接池。
    • 会话恢复：Session Ticket（无状态，服务端不存，适合集群，但要注意多机共享 ticket key 并定期轮转，否则命中率低或前向保密受损）或 Session ID（服务端存，需共享缓存）。
    • OCSP Stapling：服务端预先取好证书吊销状态一起发给客户端，省掉客户端额外去 CA 查一次（那可是一次全新的 DNS + TCP + TLS）。
    • 证书链精简：用 ECDSA 证书替代 RSA（签名短、验签快），去掉多余的中间证书，让整个链能塞进初始拥塞窗口。
    • 硬件加速：AES-NI 现在是标配（所以 AES-GCM 通常比 ChaCha20 快，移动端无 AES-NI 时反之）；超大规模才考虑 QAT 卸载或 kTLS。

    0-RTT 要谨慎：它的早期数据可被重放，只能用于幂等请求（GET），不能用来下单。

9.8.3 高级（体现深度）

  1. 高级C10K 与 C10M 分别卡在哪里？解决思路有什么本质不同？

    C10K（单机 1 万并发）卡在编程模型：一连接一线程/进程，1 万连接就是 1 万个线程，调度器、内存和上下文切换先垮。解法是换模型——非阻塞 I/O + epoll + Reactor，让少量线程（≈核数）驱动大量连接。这个问题在 2000 年代就已经被彻底解决，今天任何一个成熟框架都能轻松扛住。

    C10M（单机 1000 万）卡在内核本身。此时瓶颈已经不是"你的代码怎么写"，而是每个包都要走一遍的内核路径：协议栈处理、skb 分配、用户-内核拷贝、系统调用、软中断、锁与 cacheline 争用。哪怕每包只花 1 μs，千万级 PPS 也不够分。所以思路从"更好地使用内核"变成"绕开内核"：

    • 内核旁路：DPDK / AF_XDP 把网卡队列直接映射到用户态，轮询取包（不用中断），配用户态 TCP 栈（mTCP、F-Stack、Seastar、VPP）。
    • 零拷贝：sendfile/splice/MSG_ZEROCOPY，或干脆全程在用户态 buffer 里流转。
    • 大页：减少 TLB miss——千万级连接的状态表远超 TLB 覆盖范围。
    • share-nothing 分片：多队列网卡 + RSS 把流按哈希分到各核，每核一份独立的连接表与内存池，核间不共享任何可写状态，从根上消灭锁和 cacheline 弹跳。Seastar 是这个思想的极致体现。
    • NUMA 亲和：网卡、内存、处理线程绑在同一个 NUMA node，避免跨节点访存。

    要有判断力的地方是什么时候不该上 C10M 那套：内核旁路意味着放弃内核提供的一切——iptables、tcpdump、ss、成熟的拥塞控制、以及运维团队的全部肌肉记忆，还要独占若干个核做轮询（CPU 永远 100%）。绝大多数业务的正确答案是横向扩展加机器，而不是把单机榨到 1000 万。

  2. 高级accept 队列满了会发生什么？怎么观测和调优 backlog？

    先分清两个队列，面试里混为一谈是常见扣分点：

    • 半连接队列（SYN queue）：收到 SYN、回了 SYN-ACK、等待对方 ACK。容量受 net.ipv4.tcp_max_syn_backlog 影响。满了会丢弃新 SYN（客户端表现为连接超时）。SYN Flood 攻击打的就是这个队列，防御手段是 net.ipv4.tcp_syncookies=1——队列满时不再占用内存记录状态，而是把状态编码进 SYN-ACK 的序列号里。
    • 全连接队列（accept queue）：三次握手已完成，等着应用调 accept() 取走。容量是 min(listen() 的 backlog, net.core.somaxconn)。

    全连接队列满时的行为由 net.ipv4.tcp_abort_on_overflow 决定：默认 0，直接丢弃这个 ACK，装作没收到——服务端会按 RTO 重传 SYN-ACK，客户端以为连接已建立并开始发数据，结果长时间没有响应，最终超时。这就是为什么故障现象常常是"客户端偶发几秒超时，服务端日志里干干净净"。设成 1 则立刻回 RST，客户端快速失败（Connection reset by peer），排查更直观但对客户端不友好，一般只在定位问题时临时打开。

    观测：

    ss -lnt
State   Recv-Q  Send-Q  Local Address:Port
LISTEN  0       128     0.0.0.0:8080
# LISTEN 行的 Recv-Q = 当前全连接队列长度，Send-Q = 队列容量上限

nstat -az | grep -i listen
TcpExtListenOverflows   1832   # 全连接队列溢出次数，只要在涨就是问题
TcpExtListenDrops       1832

    调优要抓根因：ListenOverflows 增长的本质是应用 accept 得不够快，而不是队列太小。队列只是缓冲，调大它只能扛住更长的突发。真正的修复顺序是：① 确认 accept 循环没有被业务逻辑阻塞（accept 之后立刻交给 worker，绝不在 I/O 线程里做重活）；② 用 SO_REUSEPORT 多 worker 并行 accept；③ 在此基础上把 somaxconn 和 listen() 的 backlog 一起调到 1024–4096（两个都要改，只改内核参数而代码里还写着 listen(fd, 5) 是白改）。

  3. 高级服务端堆积大量 CLOSE_WAIT 说明什么？和 TIME_WAIT 的根因有什么本质区别？

    这两个状态经常被并列提起，但性质完全相反：TIME_WAIT 是协议要求你等，CLOSE_WAIT 是你自己没动。

    会自己消失吗会，60 s 后自动回收不会，永远挂着直到进程退出
    根因通常是设计问题（短连接太多）几乎一定是代码 bug

    所以看到 CLOSE_WAIT 持续增长，方向非常明确——去代码里找漏掉的 close()，调内核参数没有任何意义（内核也不该替你关，因为半关闭状态下你可能还想继续发数据）。常见的四种写法错误：

    • read() 返回 0（对端已关闭）时只 break 出循环，忘了 close(fd) 和从 epoll 里摘除。这是第一大元凶。
    • 异常/错误分支里提前 return，跳过了清理代码——用 RAII 包住 fd（unique_ptr 带 deleter 或自写 ScopedFd）可以根治。
    • 业务线程还持有该连接的引用，I/O 线程不敢关，双方互相等——引用计数或所有权设计没想清楚。
    • 连接被扔进某个 map 后异常路径没删除，对象泄漏，fd 跟着泄漏。

    定位命令：

    ss -tan state close-wait | head
lsof -p <pid> | grep -c CLOSE_WAIT
ls /proc/<pid>/fd | wc -l        # 和 ulimit -n 对照，看还有多久打满

    后果是 fd 泄漏：撑到 ulimit -n 后 accept() 开始返回 EMFILE，整个服务无法接受新连接却又不崩溃——这类"半死不活"的故障最难被健康检查发现，所以应该把 fd 使用率做成监控指标并设告警。

  4. 高级如何设计一个支撑百万长连接的服务器？

    先算账再设计——百万连接的约束条件基本由内存决定：

    • 内存：每连接的成本 = 内核 socket 结构（约 3 KB）+ 收发缓冲区 + 应用侧会话对象。收发缓冲区是最大的可调项，默认 tcp_rmem/tcp_wmem 动辄几百 KB，百万连接直接 OOM；长连接心跳场景应调小最小值（如 4 KB）并依赖内核的自动调节。整体预算 10–20 GB 是合理量级。应用侧的会话对象要精打细算，一个 std::string 的 SSO 溢出乘以 100 万就是几十 MB。
    • fd：ulimit -n 1048576 + fs.file-max，容器里还要看 LimitNOFILE。

    架构：per-core worker（进程或线程）+ SO_REUSEPORT，每个 worker 一个独立的 epoll Reactor，share-nothing——连接只属于一个 worker，避免跨核锁。I/O 线程只做收发与协议编解码，业务逻辑丢给独立线程池，绝不在 Reactor 里做任何可能阻塞的事（包括打同步日志、查数据库）。

    连接管理：连接表用开放寻址哈希表（避免每连接一次 new）；超时用时间轮而不是每连接一个定时器——百万个 std::map 定时器的插入删除会把 CPU 吃光，时间轮的每 tick 处理是 O(过期数)。

    心跳与保活：应用层心跳（30 s 级）是主力，因为它能同时探活应用是否还在响应；TCP keepalive 只做兜底（默认 2 小时太长，需调 tcp_keepalive_time）。要处理"假连接"：NAT 超时、手机切网后连接看似还在其实已死，只有主动探测能发现。

    内核参数：somaxconn、tcp_max_syn_backlog、ip_local_port_range（如果服务端也要作为客户端出连接）、netdev_max_backlog、网卡多队列 + RSS + 中断亲和。

    最容易被忽略的两件事：① 惊群式重连——一台机器挂了，它的 25 万连接会在同一秒涌向其它机器，把幸存者一并压垮。客户端重连必须做指数退避 + 随机抖动，服务端要有连接建立速率限制。② 优雅升级——百万连接全部重连一次的代价极高，需要 SO_REUSEPORT 滚动替换 worker，或用 fd 传递（SCM_RIGHTS）把连接交接给新进程。
